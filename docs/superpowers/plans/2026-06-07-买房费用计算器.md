```javascript
const {createApp} = Vue;

const app = createApp({
	data() {
		return {
			// 房源
			houseType: 'second',       // 'new' | 'second'
			totalPrice: 600,           // 万元
			area: 100,                 // ㎡

			// 资格
			homeCount: 'first',        // 'first' | 'second'
			location: 'outer',         // 'outer' (外环外) | 'inner' (外环内)

			// 二手房税费条件
			exemptVat: true,           // 满2年免征增值税
			exemptPit: false,          // 满五唯一免征个税
			buyerPayPit: true,         // 买方承担个税

			// 贷款
			loanType: 'combo',         // 'combo' (组合贷) | 'commercial' (纯商贷) | 'fund' (纯公积金)
			fundAmount: 160,           // 公积金贷款额度（万元）
			fundRate: 2.85,            // 公积金利率（%）
			commercialRate: 3.3,       // 商贷利率（%）
			loanYears: 30,             // 贷款年限
			repaymentMode: 'equal',    // 'equal' (等额本息) | 'principal' (等额本金)
		};
	},

	computed: {
		// --- 政策参数 ---

		// 首付比例
		downPaymentRatio() {
			// 首套：外环外 20%，外环内 25%
			// 二套：外环外 30%，外环内 35%
			if (this.homeCount === 'first') {
				return this.location === 'outer' ? 0.20 : 0.25;
			}
			return this.location === 'outer' ? 0.30 : 0.35;
		},

		// 契税税率
		deedTaxRate() {
			if (this.homeCount === 'first') {
				return this.area <= 90 ? 0.01 : 0.015;
			}
			// 二套
			return this.area <= 140 ? 0.02 : 0.03;
		},

		// 首付金额（万元）
		downPayment() {
			return this.totalPrice * this.downPaymentRatio;
		},

		// 契税（万元）
		deedTax() {
			return this.totalPrice * this.deedTaxRate;
		},

		// 维修基金（万元）— 仅新房
		maintenanceFund() {
			if (this.houseType === 'new') {
				return this.area * 80 / 10000; // 80元/㎡ → 万元
			}
			return 0;
		},

		// 登记费 + 印花税（万元）
		registrationFee() {
			return (80 + this.totalPrice * 10000 * 0.0005) / 10000;
		},

		// 中介费（万元）— 仅二手房
		agencyFee() {
			if (this.houseType === 'second') {
				return this.totalPrice * 0.02;
			}
			return 0;
		},

		// 增值税（万元）
		vat() {
			if (this.houseType === 'second' && !this.exemptVat) {
				return this.totalPrice * 0.053;
			}
			return 0;
		},

		// 个人所得税（万元）
		pit() {
			if (this.houseType === 'second' && !this.exemptPit && this.buyerPayPit) {
				// 普通住宅 1%，非普通 2%（用面积144㎡区分）
				const rate = this.area <= 144 ? 0.01 : 0.02;
				return this.totalPrice * rate;
			}
			return 0;
		},

		// --- 费用汇总 ---

		feeItems() {
			const items = [
				{name: '首付', value: this.downPayment},
				{name: '契税', value: this.deedTax},
			];

			if (this.houseType === 'second') {
				items.push({name: '中介费', value: this.agencyFee});
			}

			items.push(
				{name: '增值税', value: this.vat, exempt: this.houseType === 'second' && this.exemptVat},
				{name: '个人所得税', value: this.pit, exempt: this.houseType === 'second' && (this.exemptPit || !this.buyerPayPit)},
			);

			if (this.houseType === 'new') {
				items.push({name: '维修基金', value: this.maintenanceFund});
			}

			items.push({name: '登记费 + 印花税', value: this.registrationFee});

			return items;
		},

		totalCash() {
			return this.feeItems.reduce((sum, item) => sum + item.value, 0);
		},

		// --- 贷款计算 ---

		loanPrincipal() {
			// 贷款总额 = 总价 − 首付
			return this.totalPrice - this.downPayment;
		},

		fundLoanAmount() {
			if (this.loanType === 'commercial') return 0;
			if (this.loanType === 'fund') return Math.min(this.fundAmount, this.loanPrincipal());
			// combo: 公积金部分 = min(输入额度, 贷款总额)
			return Math.min(this.fundAmount, this.loanPrincipal());
		},

		commercialLoanAmount() {
			if (this.loanType === 'fund') return 0;
			if (this.loanType === 'combo') {
				return this.loanPrincipal() - this.fundLoanAmount;
			}
			return this.loanPrincipal();
		},

		// 等额本息月供公式
		equalInstallment(principal, annualRate, years) {
			if (principal <= 0 || annualRate <= 0 || years <= 0) return 0;
			const monthlyRate = annualRate / 100 / 12;
			const periods = years * 12;
			const pow = Math.pow(1 + monthlyRate, periods);
			return principal * 10000 * monthlyRate * pow / (pow - 1);
		},

		// 等额本金首月月供
		equalPrincipalFirstMonth(principal, annualRate, years) {
			if (principal <= 0 || annualRate <= 0 || years <= 0) return 0;
			const monthlyRate = annualRate / 100 / 12;
			const periods = years * 12;
			return principal * 10000 / periods + principal * 10000 * monthlyRate;
		},

		// 等额本金总利息
		equalPrincipalTotalInterest(principal, annualRate, years) {
			if (principal <= 0 || annualRate <= 0 || years <= 0) return 0;
			const monthlyRate = annualRate / 100 / 12;
			const periods = years * 12;
			// 总利息 = (首期利息 + 末期利息) × 期数 / 2
			const firstInterest = principal * 10000 * monthlyRate;
			const lastInterest = (principal * 10000 / periods) * monthlyRate;
			return (firstInterest + lastInterest) * periods / 2;
		},

		// 等额本息总利息
		equalInstallmentTotalInterest(principal, annualRate, years) {
			const monthly = this.equalInstallment(principal, annualRate, years);
			return monthly * years * 12 - principal * 10000;
		},

		// 月供（根据还款方式）
		fundMonthly() {
			if (this.repaymentMode === 'equal') {
				return this.equalInstallment(this.fundLoanAmount, this.fundRate, this.loanYears);
			}
			return this.equalPrincipalFirstMonth(this.fundLoanAmount, this.fundRate, this.loanYears);
		},

		commercialMonthly() {
			if (this.repaymentMode === 'equal') {
				return this.equalInstallment(this.commercialLoanAmount, this.commercialRate, this.loanYears);
			}
			return this.equalPrincipalFirstMonth(this.commercialLoanAmount, this.commercialRate, this.loanYears);
		},

		totalInterest() {
			let interest = 0;
			if (this.repaymentMode === 'equal') {
				interest += this.equalInstallmentTotalInterest(this.fundLoanAmount, this.fundRate, this.loanYears);
				interest += this.equalInstallmentTotalInterest(this.commercialLoanAmount, this.commercialRate, this.loanYears);
			} else {
				interest += this.equalPrincipalTotalInterest(this.fundLoanAmount, this.fundRate, this.loanYears);
				interest += this.equalPrincipalTotalInterest(this.commercialLoanAmount, this.commercialRate, this.loanYears);
			}
			return interest;
		},

		totalRepayment() {
			return this.loanPrincipal() * 10000 + this.totalInterest();
		},

		// --- 提示文本 ---
		hintText() {
			const parts = [];
			const dpPct = (this.downPaymentRatio * 100).toFixed(0);
			parts.push(`首付 ${dpPct}%`);
			const dtPct = (this.deedTaxRate * 100).toFixed(1);
			parts.push(`契税 ${dtPct}%`);
			if (this.houseType === 'second') {
				parts.push('中介费 2%');
			}
			return parts.join(' · ');
		},
	},
}).use(ElementPlus).mount('main > article');
```
