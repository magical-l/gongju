# 符号页 ZWJ 序列支持 + 全量打标 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让符号页网格支持任意长度码位序列（ZWJ 序列），并全量打标 1614 个 ZWJ 序列。

**Architecture:** 把现有双码位旗帜 seqs 机制泛化为任意长度序列：数据格式 `[cp1,...,cpN,zh,en]` 不变，前端引入 `seqCps()`（剥离尾部字符串）与整串键 `cps.join('-')`，泛化 ~15 处双码位假设；服务器 `seqs_contains`/`seqs_remove` 改精确比较；新增 `build_zwj.py` 从 emoji-test.txt 生成名字并按归属映射注入 标签.json。

**Tech Stack:** 纯 JS（Vue 3 global build）+ 纯 Python（stdlib）。无测试框架——验证用 dev_server.py + Playwright。

## Global Constraints

- 项目纯静态，**禁止 package.json / node_modules / npm 包**
- 标签.json 是唯一权威；写回 `json.dump(indent=2, ensure_ascii=False)` 文本模式（Windows 自动 CRLF），git diff 干净
- seqs 条目格式统一 `[cp1,...,cpN,zh,en]`（N≥2，码位在前 + 中英文名）；**不得**把 ZWJ 序列塞进 ranges
- 中文名肤色词（简化版）：🏻浅肤色/🏼中浅肤色/🏽中肤色/🏾中深肤色/🏿深肤色；拼接**肤色在前**（浅肤色红发男人）
- 机械轴（文字系统/官方分类/区块）禁改
- 符号.html 缓存 `?v=` 同步改（YYMMDD 惯例）

```python
def seq_cps(s):
    """seqs 条目 → 纯码位数组（剥离末尾 zh/en 字符串）。旗帜 [a,b,zh,en] 与 ZWJ [a..n,zh,en] 同构。"""
    i = len(s)
    while i > 0 and isinstance(s[i - 1], str):
        i -= 1
    return s[:i]


def seqs_contains(seqs, cps):
    return any(seq_cps(s) == cps for s in seqs)


def seqs_remove(node, cps):
    seqs = node.get('seqs', [])
    for i, s in enumerate(seqs):
        if seq_cps(s) == cps:
            seqs.pop(i)
            return True
    return False
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""从 emoji-test.txt 生成 ZWJ 序列数据并注入 标签.json。
ZWJ 序列（3+ 码位，含 U+200D）塞不进单码位 ranges，节点用 seqs（同旗帜机制）:
  seqs: [[cp1, ..., cpN, zh, en], ...]
"""
import json
import os

BASE = os.path.dirname(os.path.abspath(__file__))
EMOJI_TEST = os.path.join(BASE, '参考资料', 'emoji-test.txt')
ANNOTATIONS = os.path.join(BASE, '参考资料', 'annotations-zh.json')
TAG_FILE = os.path.join(BASE, '标签.json')

# 肤色词（用户裁定简化版；拼接肤色在前）
SKIN_ZH = {0x1F3FB: '浅肤色', 0x1F3FC: '中浅肤色', 0x1F3FD: '中肤色', 0x1F3FE: '中深肤色', 0x1F3FF: '深肤色'}


def strip_fe0f(s):
    return s.replace('️', '')


def load_cldr():
    """CLDR zh 注解：剥 VS16 后的序列串 → 中文名列表"""
    d = json.load(open(ANNOTATIONS, encoding='utf-8'))
    return {strip_fe0f(k): v['default'] for k, v in d['annotations']['annotations'].items()}


def parse_emoji_test():
    """全部含 U+200D 的 fully-qualified 序列 → [(cps, en_name), ...]"""
    out = []
    with open(EMOJI_TEST, encoding='utf-8') as f:
        for line in f:
            if line.startswith('#') or not line.strip():
                continue
            if line.split(';')[1].split()[0] != 'fully-qualified':
                continue
            cps = [int(c, 16) for c in line.split(';')[0].strip().split()]
            if 0x200D not in cps:
                continue
            out.append((cps, line.split('#')[1].strip()))
    return out


def is_skin(cp):
    return 0x1F3FB <= cp <= 0x1F3FF
```

```python
# 程序化兜底词表（CLDR 未命中的 61 个基础序列：发型/面向右/爱情/家庭）
FALLBACK_WORDS = {
    'man': '男人', 'woman': '女人', 'person': '人',
    'red hair': '红发', 'curly hair': '卷发', 'white hair': '白发', 'bald': '秃顶', 'beard': '胡须',
    'blond': '金发', 'boy': '男孩', 'girl': '女孩', 'adult': '大人', 'child': '孩子',
}


def fallback_zh(en):
    """从 emoji-test 英文名翻译。覆盖家庭/发型/爱情/面向右。"""
    # 面向右：'person walking facing right' → 基础名 + '朝右'（基础名从 CLDR 或规则出）
    m = re.search(r'(.*) facing right$', en)
    if m:
        base = m.group(1)
        zh = fallback_zh(base)
        return zh + '朝右' if zh else ''
    # 家庭：'family: man, woman, boy'
    if en.startswith('family:'):
        members = [FALLBACK_WORDS.get(x.strip(), x.strip()) for x in en.split(':', 1)[1].split(',')]
        return '家庭：' + '、'.join(members)
    # 爱情：'kiss: woman, man' / 'couple with heart: woman, man'
    m = re.match(r'kiss: (.*)$', en)
    if m:
        pair = [FALLBACK_WORDS.get(x.strip(), x.strip()) for x in m.group(1).split(',')]
        return ''.join(pair) + '接吻'
    m = re.match(r'couple with heart: (.*)$', en)
    if m:
        pair = [FALLBACK_WORDS.get(x.strip(), x.strip()) for x in m.group(1).split(',')]
        return ''.join(pair) + '情侣'
    # 发型：'man: red hair'
    m = re.match(r'(man|woman|person): (red hair|curly hair|white hair|bald|beard)$', en)
    if m:
        person, hair = m.groups()
        return FALLBACK_WORDS[hair] + FALLBACK_WORDS[person]
    # 其他未命中：退回英文名
    return ''


def zh_of(cps, en, cldr):
    """一条序列的中文名：肤色变体 = 肤色词 + 基础名；基础 = CLDR 优先，未命中程序化兜底"""
    skin = [c for c in cps if is_skin(c)]
    if skin:
        # 剥肤色得到基础序列，求基础名
        base_cps = [c for c in cps if not is_skin(c)]
        base_zh = zh_of(base_cps, en, cldr)
        return SKIN_ZH[skin[0]] + base_zh
    key = strip_fe0f(''.join(chr(c) for c in cps))
    if key in cldr:
        return cldr[key][0]
    return fallback_zh(en)
```

            node = get_node(roots, path)
            if node is None:
                print(f'!! 标签不存在: {path}')

            node.setdefault('seqs', [])
            seqs_existing = node['seqs']
            entry = cps + [zh, en]
            if not any(seq_cps_same(s, cps) for s in seqs_existing):
                seqs_existing.append(entry)
                added[path] = added.get(path, 0) + 1
    # 排序（按码位）

        node = get_node(roots, path)
        node['seqs'].sort(key=lambda s: tuple(s[:2]))
    with open(TAG_FILE, 'w', encoding='utf-8', newline='') as f:

def get_node(roots, path):
    parts = path.split('/')
    node = roots.get(parts[0])
    for p in parts[1:]:
        node = node.get('children', {}).get(p)
        if node is None:
            return None
    return node


def seq_cps_same(s, cps):
    i = len(s)
    while i > 0 and isinstance(s[i - 1], str):
        i -= 1
    return s[:i] == cps


if __name__ == '__main__':
    main()
