```python
import json

def rebuild_ordered(d, key, after=None):
    """保序重排：把 key 移到 after 之后（after=None 移到最前）。"""
    items = list(d.items())
    entry = dict([(k, v) for k, v in items if k == key])
    rest = [(k, v) for k, v in items if k != key]
    out = {}
    for k, v in rest:
        out[k] = v
        if after is not None and k == after:
            out.update(entry)
    if after is None:
        out = {**entry, **out}
    if key not in out:  # after 未命中则追加到末尾
        out[key] = d[key]
    return out

data = json.load(open('标签.json', encoding='utf-8'))
roots = data['roots']
r = roots.pop('人与身体')          # 从根集合取出并删除
kids = r['children']

# --- 人：通用人取消 → 本级；其余 6 节点迁入 ---
ren_children = {}
for name in ['人物角色', '人形图示、火柴人', '性别', '未成年人', '衰老、老人', '死亡、致命']:
    ren_children[name] = kids.pop(name)
clothing = kids.pop('仪容仪表')                      # → 服饰
cosmetics = clothing['children'].pop('梳妆打扮')     # 💄 → 人
ren_children['梳妆打扮'] = cosmetics
tongyong = kids.pop('通用人')                        # 取消：字符升为 人 本级
ren = {'children': ren_children,
       'ranges': tongyong.get('ranges', []),
       'seqs': tongyong.get('seqs', []),
       'src': tongyong.get('src', [])}
if tongyong.get('alias'): ren['alias'] = tongyong['alias']

# --- 身体部位：本级6 + 原6子 + 脸 + 发型发色 ---
body = kids.pop('身体部位')
body['children']['脸'] = kids.pop('脸')
body['children']['发型、发色'] = kids.pop('发型、发色')

# --- 表情、表达：原「表达」提升改名 ---
expr = kids.pop('表达')
expr = {'表情、表达': expr}

# --- 服饰：仪容仪表改名（梳妆打扮已迁出） ---
fushi = {'children': clothing['children'],
         'ranges': clothing.get('ranges', []),
         'seqs': clothing.get('seqs', []),
         'alias': clothing.get('alias'),
         'src': clothing.get('src', [])}

# --- 健康并入社会生活（插到 教育、学习、书本 之后） ---
health = kids.pop('健康、疾病、治疗')
social = roots['社会生活']['children']
social = rebuild_ordered(social, '健康、疾病、治疗', after='教育、学习、书本')
roots['社会生活']['children'] = social

assert not kids, f'人与身体残留未处理: {list(kids.keys())}'

# --- 新根按序插入（占原 人与身体 位置，即 区块 之后） ---
roots = rebuild_ordered(roots, '服饰', after='表情、表达')  # 先按依赖序逐键插
# 更稳做法：整体重建前四键顺序
order = []
for k in ['文字系统', '官方分类', '区块']:
    order.append((k, roots.pop(k)))
order += [('人', ren), ('身体部位', body)]
order += list(expr.items())
order += [('服饰', fushi)]
new_roots = dict(order)
for k, v in roots.items():
    new_roots[k] = v
roots.clear(); roots.update(new_roots)

with open('标签.json', 'w', encoding='utf-8') as f:
    json.dump(data, f, indent=2, ensure_ascii=False)
print('迁移完成')
```
