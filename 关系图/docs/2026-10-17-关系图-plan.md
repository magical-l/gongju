```js
    fn();

  } catch (err) {
    fail += 1;

    console.log('  ✗ ' + name);

function assertEqual(actual, expected, message) {
  if (actual !== expected) {
    throw new Error((message || '值不相等') + '：期望 ' + JSON.stringify(expected) + '，实际 ' + JSON.stringify(actual));
  }
}

function assertDeepEqual(actual, expected, message) {
  const a = JSON.stringify(actual);
  const b = JSON.stringify(expected);
  if (a !== b) throw new Error((message || '结构不相等') + '：期望 ' + b + '，实际 ' + a);
}

function assertTruthy(value, message) {
  if (!value) throw new Error(message || '期望为真');
}

function assertThrows(fn, includes, message) {
  try {
    fn();
  } catch (err) {
    if (includes && String(err.message).indexOf(includes) === -1) {
      throw new Error((message || '异常信息不符') + '：期望包含「' + includes + '」，实际「' + err.message + '」');
    }
    return;
  }
  throw new Error(message || '期望抛出异常');
}

function summary() {
  console.log('\n通过 ' + pass + '，失败 ' + fail);
```

```js
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory();
  else root.RGModel = factory();
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';

  const SCHEMA_VERSION = 1;
  const PORTS = ['top', 'right', 'bottom', 'left'];
  const ENTITY_STYLE_DEFAULTS = Object.freeze({
    shape: 'rounded', fill: '#ffffff', border: '#9aa3b2', text: '#1f2430', width: 168
  });
  const RELATION_STYLE_DEFAULTS = Object.freeze({
    color: '#5b6472', dash: false, arrow: 'end', lineStyle: 'curve', showLabel: true, width: 2
  });
  const ENTITY_STYLE_KEYS = ['shape', 'fill', 'border', 'text', 'width'];
  const RELATION_STYLE_KEYS = ['color', 'dash', 'arrow', 'lineStyle', 'showLabel', 'width'];
  const SHAPES = ['rounded', 'rect', 'pill'];
  const ARROWS = ['end', 'both', 'none'];
  const LINE_STYLES = ['curve', 'straight'];

  let idCounter = 0;
  function newId(prefix) {
    idCounter += 1;
    return prefix + Date.now().toString(36) + '_' + idCounter.toString(36) + Math.random().toString(36).slice(2, 6);
  }

  function kindOf(obj) {
    if (!obj || typeof obj.id !== 'string') return null;
    if (obj.id.startsWith('n_')) return 'entity';
    if (obj.id.startsWith('r_')) return 'relation';
    return null;
  }

  function createDocument(name) {
    return {
      schemaVersion: SCHEMA_VERSION,
      name: name || '未命名关系图',
      entities: [],
      relations: [],
      viewport: { x: 0, y: 0, zoom: 1 }
    };
  }

  function findEntity(doc, id) {
    return doc.entities.find(function (e) { return e.id === id; }) || null;
  }

  function findRelation(doc, id) {
    return doc.relations.find(function (r) { return r.id === id; }) || null;
  }

  function findObject(doc, id) {
    return findEntity(doc, id) || findRelation(doc, id);
  }

  function getParent(doc, obj) {
    if (!obj || !obj.prototypeId) return null;
    return kindOf(obj) === 'entity' ? findEntity(doc, obj.prototypeId) : findRelation(doc, obj.prototypeId);
  }

  function getChildren(doc, obj) {
    const arr = kindOf(obj) === 'entity' ? doc.entities : doc.relations;
    return arr.filter(function (o) { return o.prototypeId === obj.id && o.id !== obj.id; });
  }

  function getRoots(doc, kind) {
    const arr = kind === 'entity' ? doc.entities : doc.relations;
    return arr.filter(function (o) { return !o.prototypeId; });
  }

  function getDescendantIds(doc, obj) {
    const out = new Set();
    const walk = function (node) {
      getChildren(doc, node).forEach(function (child) {
        if (out.has(child.id)) return;
        out.add(child.id);
        walk(child);
      });
    };
    walk(obj);
    return out;
  }

  function isDescendant(doc, candidateId, ancestorId) {
    const ancestor = findObject(doc, ancestorId);
    return !!ancestor && getDescendantIds(doc, ancestor).has(candidateId);
  }

  function createEntity(doc, data) {
    const d = data || {};
    const id = d.id || newId('n_');
    if (findObject(doc, id)) throw new Error('ID 已存在: ' + id);
    if (d.prototypeId && !findEntity(doc, d.prototypeId)) throw new Error('父原型不存在: ' + d.prototypeId);
    const entity = {
      id: id,
      name: typeof d.name === 'string' && d.name ? d.name : '新实体',
      prototypeId: d.prototypeId || null,
      position: d.position || null,
      style: d.style ? Object.assign({}, d.style) : {}
    };
    doc.entities.push(entity);
    return entity;
  }

  function createRelation(doc, data) {
    const d = data || {};
    const id = d.id || newId('r_');
    if (findObject(doc, id)) throw new Error('ID 已存在: ' + id);
    if (d.prototypeId && !findRelation(doc, d.prototypeId)) throw new Error('父原型不存在: ' + d.prototypeId);
    const relation = {
      id: id,
      name: typeof d.name === 'string' ? d.name : '',
      prototypeId: d.prototypeId || null,
      ends: d.ends || null,
      style: d.style ? Object.assign({}, d.style) : {}
    };
    doc.relations.push(relation);
    return relation;
  }

  return {
    SCHEMA_VERSION: SCHEMA_VERSION,
    PORTS: PORTS,
    ENTITY_STYLE_DEFAULTS: ENTITY_STYLE_DEFAULTS,
    RELATION_STYLE_DEFAULTS: RELATION_STYLE_DEFAULTS,
    ENTITY_STYLE_KEYS: ENTITY_STYLE_KEYS,
    RELATION_STYLE_KEYS: RELATION_STYLE_KEYS,
    SHAPES: SHAPES,
    ARROWS: ARROWS,
    LINE_STYLES: LINE_STYLES,
    kindOf: kindOf,
    createDocument: createDocument,
    findEntity: findEntity,
    findRelation: findRelation,
    findObject: findObject,
    getParent: getParent,
    getChildren: getChildren,
    getRoots: getRoots,
    getDescendantIds: getDescendantIds,
    isDescendant: isDescendant,
    createEntity: createEntity,
    createRelation: createRelation
  };
});
```

```js
  function setParent(doc, obj, parentId) {
    if (parentId === obj.id) throw new Error('不能以自己为原型');
    if (parentId) {
      const parent = kindOf(obj) === 'entity' ? findEntity(doc, parentId) : findRelation(doc, parentId);
      if (!parent) throw new Error('父原型不存在');
      if (isDescendant(doc, parentId, obj.id)) throw new Error('不能挂到自己的后代下');
    }
    obj.prototypeId = parentId || null;
  }

  function styleDefaults(kind) {
    return kind === 'entity' ? ENTITY_STYLE_DEFAULTS : RELATION_STYLE_DEFAULTS;
  }

  function styleChain(doc, obj) {
    const chain = [];
    const seen = new Set();
    let cur = obj;
    while (cur && !seen.has(cur.id)) {
      seen.add(cur.id);
      chain.unshift(cur);
      cur = getParent(doc, cur);
    }
    return chain;
  }

  function resolveStyle(doc, obj) {
    const style = Object.assign({}, styleDefaults(kindOf(obj)));
    styleChain(doc, obj).forEach(function (node) {
      Object.assign(style, node.style || {});
    });
    return style;
  }

  function getStyleSource(doc, obj, key) {
    const chain = styleChain(doc, obj);
    let found = null;
    for (let i = chain.length - 1; i >= 0; i -= 1) {
      const node = chain[i];
      if (node.style && Object.prototype.hasOwnProperty.call(node.style, key)) {
        found = node;
        break;
      }
    }
    if (!found) return { value: styleDefaults(kindOf(obj))[key], source: 'default' };
    const sourceName = kindOf(found) === 'entity' ? found.name : effectiveRelationLabel(doc, found);
    return {
      value: found.style[key],
      source: found === obj ? 'own' : 'inherited',
      sourceId: found.id,
      sourceName: sourceName
    };
  }

  function setStyleKey(obj, key, value) {
    obj.style = obj.style || {};
    obj.style[key] = value;
  }

  function clearStyleKey(obj, key) {
    if (obj.style) delete obj.style[key];
  }

  function effectiveRelationLabel(doc, relation) {
    const seen = new Set();
    let cur = relation;
    while (cur && !seen.has(cur.id)) {
      if (typeof cur.name === 'string' && cur.name.trim()) return cur.name.trim();
      seen.add(cur.id);
      cur = getParent(doc, cur);
    }
    return '关系';
  }

  function getLabelSource(doc, relation) {
    const seen = new Set();
    let cur = relation;
    while (cur && !seen.has(cur.id)) {
      if (typeof cur.name === 'string' && cur.name.trim()) {
        if (cur === relation) return { value: cur.name.trim(), source: 'own' };
        return { value: cur.name.trim(), source: 'inherited', sourceId: cur.id, sourceName: cur.name.trim() };
      }
      seen.add(cur.id);
      cur = getParent(doc, cur);
    }
    return { value: '关系', source: 'default' };
  }

  function isEndBound(end) {
    return !!(end && typeof end.entityId === 'string' && end.entityId);
  }

  function isEndFree(end) {
    return !!(end && Number.isFinite(end.x) && Number.isFinite(end.y));
  }

  function makeBoundEnd(entityId, port) { return { entityId: entityId, port: port }; }
  function makeFreeEnd(x, y) { return { x: x, y: y }; }

  function relationEndsAreFree(rel) {
    return !!rel.ends && isEndFree(rel.ends.a) && isEndFree(rel.ends.b);
  }

  function relationEndsAreBound(rel) {
    return !!rel.ends && isEndBound(rel.ends.a) && isEndBound(rel.ends.b);
  }

  function relationIsHalfBound(rel) {
    if (!rel.ends) return false;
    return isEndBound(rel.ends.a) !== isEndBound(rel.ends.b);
  }

  function relationIsPlaced(rel) { return !!rel.ends; }

  function relationVisibility(doc, rel) {
    if (!rel.ends) return false;
    const ends = [rel.ends.a, rel.ends.b];
    for (let i = 0; i < ends.length; i += 1) {
      if (!isEndBound(ends[i])) continue;
      const entity = findEntity(doc, ends[i].entityId);
      if (!entity || !entity.position) return false;
    }
    return true;
  }

  function placeEntity(entity, x, y) { entity.position = { x: x, y: y }; }
  function unplaceEntity(entity) { entity.position = null; }
```

```js
  function endsBoundTo(rel, entityId) {
    if (!rel.ends) return false;
    return (isEndBound(rel.ends.a) && rel.ends.a.entityId === entityId) ||
      (isEndBound(rel.ends.b) && rel.ends.b.entityId === entityId);
  }

  function deletionCheck(doc, obj) {
    const kind = kindOf(obj);
    const arr = kind === 'entity' ? doc.entities : doc.relations;
    const refs = arr.filter(function (o) { return o.prototypeId === obj.id; });
    if (refs.length) return { ok: false, reason: '有 ' + refs.length + ' 个对象以它为原型' };
    if (kind === 'relation') return { ok: true, incidentRelations: [] };
    const incident = doc.relations.filter(function (r) { return endsBoundTo(r, obj.id); });
    for (let i = 0; i < incident.length; i += 1) {
      const rel = incident[i];
      const relRefs = doc.relations.filter(function (r) { return r.prototypeId === rel.id; });
      if (relRefs.length) {
        return {
          ok: false,
          reason: '关联关系「' + effectiveRelationLabel(doc, rel) + '」仍被 ' + relRefs.length + ' 条关系引用'
        };
      }
    }
    return { ok: true, incidentRelations: incident };
  }

  function deleteObject(doc, obj) {
    const check = deletionCheck(doc, obj);
    if (!check.ok) throw new Error(check.reason);
    if (kindOf(obj) === 'entity') {
      const ids = new Set(check.incidentRelations.map(function (r) { return r.id; }));
      doc.relations = doc.relations.filter(function (r) { return !ids.has(r.id); });
      doc.entities = doc.entities.filter(function (e) { return e.id !== obj.id; });
      return { deletedRelations: ids.size };
    }
    doc.relations = doc.relations.filter(function (r) { return r.id !== obj.id; });
    return { deletedRelations: 1 };
  }

  function cloneDocument(doc) {
    return JSON.parse(JSON.stringify(doc));
  }

  function isPlainObject(value) {
    return !!value && typeof value === 'object' && !Array.isArray(value);
  }

  function validateEnd(end, entityIds) {
    const hasEntity = Object.prototype.hasOwnProperty.call(end, 'entityId');
    const hasCoord = Object.prototype.hasOwnProperty.call(end, 'x') ||
      Object.prototype.hasOwnProperty.call(end, 'y');
    if (hasEntity && hasCoord) return '端点不能同时有实体和坐标';
    if (hasEntity) {
      if (!PORTS.includes(end.port)) return 'port 非法';
      if (!entityIds.has(end.entityId)) return '实体不存在';
      return '';
    }
    if (Number.isFinite(end.x) && Number.isFinite(end.y)) return '';
    return '必须是实体端点或自由坐标';
  }

  function validateStyle(style, allowed) {
    const keys = Object.keys(style);
    for (let i = 0; i < keys.length; i += 1) {
      const key = keys[i];
      const value = style[key];
      if (!allowed.includes(key)) return '未知属性 ' + key;
      if (key === 'shape' && !SHAPES.includes(value)) return 'shape 非法';
      if (key === 'arrow' && !ARROWS.includes(value)) return 'arrow 非法';
      if (key === 'lineStyle' && !LINE_STYLES.includes(value)) return 'lineStyle 非法';
      if ((key === 'dash' || key === 'showLabel') && typeof value !== 'boolean') return key + ' 必须是布尔';
      if ((key === 'fill' || key === 'border' || key === 'text' || key === 'color') &&
        (typeof value !== 'string' || !value.trim())) return key + ' 必须是非空字符串';
      if (key === 'width' && (!Number.isFinite(value) || value <= 0)) return 'width 必须是正数';
    }
    return '';
  }

  function hasCycle(items) {
    const byId = new Map(items.map(function (o) { return [o.id, o]; }));
    for (let i = 0; i < items.length; i += 1) {
      const seen = new Set();
      let cur = items[i];
      while (cur && cur.prototypeId) {
        if (seen.has(cur.id)) return true;
        seen.add(cur.id);
        cur = byId.get(cur.prototypeId);
      }
    }
    return false;
  }

    if (!isPlainObject(raw)) return fail('文档不是对象');
    if (raw.schemaVersion !== SCHEMA_VERSION) return fail('schemaVersion 必须为 1');
    if (typeof raw.name !== 'string') return fail('name 必须是字符串');
    if (!Array.isArray(raw.entities)) return fail('entities 必须是数组');
    if (!Array.isArray(raw.relations)) return fail('relations 必须是数组');

    const ids = new Set();
    const entityIds = new Set();
    for (let i = 0; i < raw.entities.length; i += 1) {
      const e = raw.entities[i];
      const p = 'entities[' + i + ']';
      if (!isPlainObject(e)) return fail(p + ' 不是对象');
      if (typeof e.id !== 'string' || !e.id.startsWith('n_')) return fail(p + '.id 非法');
      if (ids.has(e.id)) return fail('ID 重复: ' + e.id);
      ids.add(e.id);
      entityIds.add(e.id);
      if (typeof e.name !== 'string' || !e.name.trim()) return fail(p + '.name 必须是非空字符串');
      if (e.prototypeId != null && typeof e.prototypeId !== 'string') return fail(p + '.prototypeId 非法');
      if (e.position != null) {
        if (!isPlainObject(e.position) || !Number.isFinite(e.position.x) || !Number.isFinite(e.position.y)) {
          return fail(p + '.position 非法');
        }
      }
      if (!isPlainObject(e.style)) return fail(p + '.style 必须是对象');
      const styleErr = validateStyle(e.style, ENTITY_STYLE_KEYS);
      if (styleErr) return fail(p + '.style: ' + styleErr);
    }

    const relationIds = new Set();
    for (let i = 0; i < raw.relations.length; i += 1) {
      const r = raw.relations[i];
      const p = 'relations[' + i + ']';
      if (!isPlainObject(r)) return fail(p + ' 不是对象');
      if (typeof r.id !== 'string' || !r.id.startsWith('r_')) return fail(p + '.id 非法');
      if (ids.has(r.id)) return fail('ID 重复: ' + r.id);
      ids.add(r.id);
      relationIds.add(r.id);
      if (typeof r.name !== 'string') return fail(p + '.name 必须是字符串');
      if (r.prototypeId != null && typeof r.prototypeId !== 'string') return fail(p + '.prototypeId 非法');
      if (!isPlainObject(r.style)) return fail(p + '.style 必须是对象');
      const styleErr = validateStyle(r.style, RELATION_STYLE_KEYS);
      if (styleErr) return fail(p + '.style: ' + styleErr);
      if (r.ends != null) {
        if (!isPlainObject(r.ends) || !isPlainObject(r.ends.a) || !isPlainObject(r.ends.b)) {
          return fail(p + '.ends 非法');
        }
        const errA = validateEnd(r.ends.a, entityIds);
        if (errA) return fail(p + '.ends.a: ' + errA);
        const errB = validateEnd(r.ends.b, entityIds);
        if (errB) return fail(p + '.ends.b: ' + errB);
      }
    }

    for (let i = 0; i < raw.entities.length; i += 1) {
      const e = raw.entities[i];
      if (e.prototypeId != null && !entityIds.has(e.prototypeId)) {
        return fail('实体 ' + e.id + ' 的原型不存在');
      }
    }
    for (let i = 0; i < raw.relations.length; i += 1) {
      const r = raw.relations[i];
      if (r.prototypeId != null && !relationIds.has(r.prototypeId)) {
        return fail('关系 ' + r.id + ' 的原型不存在');

    }
    if (hasCycle(raw.entities)) return fail('实体原型链存在环');
    if (hasCycle(raw.relations)) return fail('关系原型链存在环');

    const vp = raw.viewport;
    if (!isPlainObject(vp) || !Number.isFinite(vp.x) || !Number.isFinite(vp.y) || !Number.isFinite(vp.zoom)) {
      return fail('viewport 非法');
    }
    if (vp.zoom < 0.25 || vp.zoom > 3) return fail('viewport.zoom 超出范围');

    return { ok: true, doc: cloneDocument(raw) };
  }
```

```js
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory();
  else root.RGGeometry = factory();
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';

  const PORT_NORMALS = {
    top: { nx: 0, ny: -1 },
    right: { nx: 1, ny: 0 },
    bottom: { nx: 0, ny: 1 },
    left: { nx: -1, ny: 0 }
  };

  function portPoint(rect, port) {
    const n = PORT_NORMALS[port];
    if (!n) throw new Error('未知端口: ' + port);
    let x = rect.x + rect.width / 2;
    let y = rect.y + rect.height / 2;
    if (port === 'right') x = rect.x + rect.width;
    if (port === 'left') x = rect.x;
    if (port === 'top') y = rect.y;
    if (port === 'bottom') y = rect.y + rect.height;
    return { x: x, y: y, nx: n.nx, ny: n.ny };
  }

  function resolveEnd(end, getBoundPoint) {
    if (end && typeof end.entityId === 'string') {
      return getBoundPoint(end.entityId, end.port) || null;
    }
    if (end && Number.isFinite(end.x) && Number.isFinite(end.y)) {
      return { x: end.x, y: end.y, nx: null, ny: null };
    }
    return null;
  }

  function withDirections(a, b) {
    const pa = { x: a.x, y: a.y, nx: a.nx, ny: a.ny };
    const pb = { x: b.x, y: b.y, nx: b.nx, ny: b.ny };
    if (pa.nx == null || pa.ny == null) {
      const dx = pb.x - pa.x;
      const dy = pb.y - pa.y;
      const len = Math.hypot(dx, dy) || 1;
      pa.nx = dx / len;
      pa.ny = dy / len;
    }
    if (pb.nx == null || pb.ny == null) {
      const dx = pa.x - pb.x;
      const dy = pa.y - pb.y;
      const len = Math.hypot(dx, dy) || 1;
      pb.nx = dx / len;
      pb.ny = dy / len;
    }
    return [pa, pb];
  }

  function cubicPoint(p0, p1, p2, p3, t) {
    const u = 1 - t;
    return {
      x: u * u * u * p0.x + 3 * u * u * t * p1.x + 3 * u * t * t * p2.x + t * t * t * p3.x,
      y: u * u * u * p0.y + 3 * u * u * t * p1.y + 3 * u * t * t * p2.y + t * t * t * p3.y
    };
  }

  function buildRelationPath(endA, endB, lineStyle) {
    if (!endA || !endB) return null;
    const pair = withDirections(endA, endB);
    const a = pair[0];
    const b = pair[1];
    const dist = Math.hypot(b.x - a.x, b.y - a.y);
    if (lineStyle === 'straight') {
      return {
        d: 'M ' + a.x + ' ' + a.y + ' L ' + b.x + ' ' + b.y,
        mid: { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 },
        a: a,
        b: b
      };
    }
    const k = Math.max(40, dist * 0.4);
    const c1 = { x: a.x + a.nx * k, y: a.y + a.ny * k };
    const c2 = { x: b.x + b.nx * k, y: b.y + b.ny * k };
    return {
      d: 'M ' + a.x + ' ' + a.y + ' C ' + c1.x + ' ' + c1.y + ' ' + c2.x + ' ' + c2.y + ' ' + b.x + ' ' + b.y,
      mid: cubicPoint(a, c1, c2, b, 0.5),
      a: a,
      b: b,
      c1: c1,
      c2: c2
    };
  }

  function rectLines(rect) {
    return {
      x: [rect.x, rect.x + rect.width / 2, rect.x + rect.width],
      y: [rect.y, rect.y + rect.height / 2, rect.y + rect.height]
    };
  }

  function bestSnap(values, candidates, threshold) {
    let best = null;
    for (let i = 0; i < values.length; i += 1) {
      for (let j = 0; j < candidates.length; j += 1) {
        const diff = candidates[j] - values[i];
        if (Math.abs(diff) <= threshold && (!best || Math.abs(diff) < Math.abs(best.diff))) {
          best = { diff: diff, line: candidates[j] };
        }
      }
    }
    return best;
  }

  function snapGuides(movingRect, otherRects, threshold) {
    const moving = rectLines(movingRect);
    const otherX = [];
    const otherY = [];
    otherRects.forEach(function (rect) {
      const lines = rectLines(rect);
      otherX.push.apply(otherX, lines.x);
      otherY.push.apply(otherY, lines.y);
    });
    const sx = bestSnap(moving.x, otherX, threshold);
    const sy = bestSnap(moving.y, otherY, threshold);
    return {
      dx: sx ? sx.diff : 0,
      dy: sy ? sy.diff : 0,
      guideX: sx ? sx.line : null,
      guideY: sy ? sy.line : null
    };
  }

  function bboxOfRects(rects) {
    if (!rects.length) return null;
    let minX = Infinity;
    let minY = Infinity;
    let maxX = -Infinity;
    let maxY = -Infinity;
    rects.forEach(function (r) {
      minX = Math.min(minX, r.x);
      minY = Math.min(minY, r.y);
      maxX = Math.max(maxX, r.x + r.width);
      maxY = Math.max(maxY, r.y + r.height);
    });
    return { x: minX, y: minY, width: maxX - minX, height: maxY - minY };
  }

  function bboxOfPoints(points) {
    if (!points.length) return null;
    let minX = Infinity;
    let minY = Infinity;
    let maxX = -Infinity;
    let maxY = -Infinity;
    points.forEach(function (p) {
      minX = Math.min(minX, p.x);
      minY = Math.min(minY, p.y);
      maxX = Math.max(maxX, p.x);
      maxY = Math.max(maxY, p.y);
    });
    return { x: minX, y: minY, width: maxX - minX, height: maxY - minY };
  }

  function fitView(bbox, viewportSize, padding) {
    if (!bbox) return { x: 0, y: 0, zoom: 1 };
    const w = bbox.width + padding * 2;
    const h = bbox.height + padding * 2;
    const zoom = Math.max(0.25, Math.min(3, Math.min(viewportSize.width / w, viewportSize.height / h)));
    return {
      x: viewportSize.width / 2 - (bbox.x + bbox.width / 2) * zoom,
      y: viewportSize.height / 2 - (bbox.y + bbox.height / 2) * zoom,
      zoom: zoom
    };
  }

  function screenToWorld(clientX, clientY, viewport, originRect) {
    return {
      x: (clientX - originRect.left - viewport.x) / viewport.zoom,
      y: (clientY - originRect.top - viewport.y) / viewport.zoom
    };
  }

  function worldToScreen(point, viewport, originRect) {
    return {
      x: originRect.left + viewport.x + point.x * viewport.zoom,
      y: originRect.top + viewport.y + point.y * viewport.zoom
    };
  }

  return {
    portPoint: portPoint,
    resolveEnd: resolveEnd,
    buildRelationPath: buildRelationPath,
    cubicPoint: cubicPoint,
    snapGuides: snapGuides,
    bboxOfRects: bboxOfRects,
    bboxOfPoints: bboxOfPoints,
    fitView: fitView,
    screenToWorld: screenToWorld,
    worldToScreen: worldToScreen
  };
});
```

```js
    this.overlayLayer = canvasEl.querySelector('.rg-overlay-layer');
    this.nodeLayer = canvasEl.querySelector('.rg-node-layer');
    this.emptyEl = canvasEl.querySelector('.rg-empty');
    this.viewport = { x: 0, y: 0, zoom: 1 };
    this.doc = null;
    this.state = null;
    this.markers = new Map();
    this.bindEvents();
  }

  CanvasView.prototype.bindEvents = function () {
    const self = this;
    this.el.addEventListener('pointerdown', function (e) {
      const handle = e.target.closest('.rg-end-handle');
      if (handle) {
        e.preventDefault();
        self.cb.onHandlePointerDown && self.cb.onHandlePointerDown(handle.dataset.id, handle.dataset.end, e);
        return;
      }
      const port = e.target.closest('.rg-port');
      if (port) {
        e.preventDefault();
        const card = port.closest('.rg-card');
        self.cb.onPortPointerDown && self.cb.onPortPointerDown(card.dataset.id, port.dataset.port, e);
        return;
      }
      const card = e.target.closest('.rg-card');
      if (card) {
        e.preventDefault();
        self.cb.onCardPointerDown && self.cb.onCardPointerDown(card.dataset.id, e);
        return;
      }
      const edge = e.target.closest('[data-edge-id]');
      if (edge) {
        e.preventDefault();
        self.cb.onEdgePointerDown && self.cb.onEdgePointerDown(edge.dataset.edgeId, e);
        return;
      }
      self.cb.onCanvasPointerDown && self.cb.onCanvasPointerDown(e);
    });
    this.el.addEventListener('dblclick', function (e) {
      const card = e.target.closest('.rg-card');
      if (card) self.cb.onCardDoubleClick && self.cb.onCardDoubleClick(card.dataset.id);
    });
  };

  CanvasView.prototype.setViewport = function (vp) {
    this.viewport = { x: vp.x, y: vp.y, zoom: vp.zoom };
    this.world.style.transform = 'translate(' + vp.x + 'px,' + vp.y + 'px) scale(' + vp.zoom + ')';
    this.el.style.backgroundSize = (20 * vp.zoom) + 'px ' + (20 * vp.zoom) + 'px';
    this.el.style.backgroundPosition = vp.x + 'px ' + vp.y + 'px';
  };

  CanvasView.prototype.render = function (doc, state) {
    this.doc = doc;
    this.state = state || {};
    this.nodeLayer.textContent = '';
    this.edgeLayer.textContent = '';
    this.labelLayer.textContent = '';
    this.overlayLayer.textContent = '';
    this.renderNodes(doc);
    this.renderEdges(doc, this.state);
  };

  CanvasView.prototype.renderNodes = function (doc) {
    const self = this;
    doc.entities.forEach(function (entity) {
      if (!entity.position) return;
      const style = M.resolveStyle(doc, entity);
      const card = document.createElement('div');
      card.className = 'rg-card shape-' + style.shape;
      card.dataset.id = entity.id;
      card.style.left = entity.position.x + 'px';
      card.style.top = entity.position.y + 'px';
      card.style.width = style.width + 'px';
      card.style.setProperty('--rg-fill', style.fill);
      card.style.setProperty('--rg-border', style.border);
      card.style.setProperty('--rg-text', style.text);
      if (self.state.selectedId === entity.id) card.classList.add('selected');
      const name = document.createElement('span');
      name.className = 'rg-card-name';
      name.textContent = entity.name;
      card.appendChild(name);
      ['top', 'right', 'bottom', 'left'].forEach(function (port) {
        const dot = document.createElement('span');
        dot.className = 'rg-port';
        dot.dataset.port = port;
        card.appendChild(dot);
      });
      self.nodeLayer.appendChild(card);
    });
  };

  CanvasView.prototype.getEntityRect = function (id) {
    const card = this.nodeLayer.querySelector('.rg-card[data-id="' + id + '"]');
    if (!card) return null;
    return {
      x: card.offsetLeft,
      y: card.offsetTop,
      width: card.offsetWidth,
      height: card.offsetHeight
    };
  };

  CanvasView.prototype.resolveEnd = function (end) {
    const self = this;
    return G.resolveEnd(end, function (entityId, port) {
      const rect = self.getEntityRect(entityId);
      return rect ? G.portPoint(rect, port) : null;
    });
  };

  CanvasView.prototype.renderEdges = function (doc, state) {
    const self = this;
    const contentRects = [];
    const contentPoints = [];
    doc.entities.forEach(function (entity) {
      if (!entity.position) return;
      const rect = self.getEntityRect(entity.id);
      if (rect) contentRects.push(rect);
    });
    let hasEdges = false;
    doc.relations.forEach(function (rel) {
      if (!M.relationVisibility(doc, rel)) return;
      if (M.isEndFree(rel.ends.a)) contentPoints.push(rel.ends.a);
      if (M.isEndFree(rel.ends.b)) contentPoints.push(rel.ends.b);
      const style = M.resolveStyle(doc, rel);
      const a = self.resolveEnd(rel.ends.a);
      const b = self.resolveEnd(rel.ends.b);
      if (!a || !b) return;
      const path = G.buildRelationPath(a, b, style.lineStyle);
      if (!path) return;
      hasEdges = true;
      const hit = document.createElementNS('http://www.w3.org/2000/svg', 'path');
      hit.setAttribute('d', path.d);
      hit.setAttribute('class', 'rg-edge-hit');
      hit.setAttribute('stroke', 'transparent');
      hit.setAttribute('stroke-width', '12');
      hit.setAttribute('fill', 'none');
      hit.dataset.edgeId = rel.id;
      self.edgeLayer.appendChild(hit);

      const line = document.createElementNS('http://www.w3.org/2000/svg', 'path');
      line.setAttribute('d', path.d);
      line.setAttribute('class', 'rg-edge' + (state.selectedId === rel.id ? ' selected' : ''));
      line.dataset.edgeId = rel.id;
      line.setAttribute('stroke', style.color);
      line.setAttribute('stroke-width', String(style.width));
      if (style.dash) line.setAttribute('stroke-dasharray', '6 4');
      if (style.arrow === 'end' || style.arrow === 'both') {
        line.setAttribute('marker-end', 'url(#' + self.ensureMarker(style.color) + ')');
      }
      if (style.arrow === 'both') {
        line.setAttribute('marker-start', 'url(#' + self.ensureMarker(style.color) + ')');
      }
      self.edgeLayer.appendChild(line);

      if (style.showLabel) {
        const label = document.createElementNS('http://www.w3.org/2000/svg', 'g');
        label.setAttribute('class', 'rg-edge-label');
        label.dataset.edgeId = rel.id;
        const text = document.createElementNS('http://www.w3.org/2000/svg', 'text');
        text.setAttribute('x', String(path.mid.x));
        text.setAttribute('y', String(path.mid.y));
        text.setAttribute('text-anchor', 'middle');
        text.setAttribute('dominant-baseline', 'middle');
        text.textContent = M.effectiveRelationLabel(doc, rel);
        label.appendChild(text);
        self.labelLayer.appendChild(label);
        const box = text.getBBox();
        const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
        rect.setAttribute('x', String(box.x - 4));
        rect.setAttribute('y', String(box.y - 2));
        rect.setAttribute('width', String(box.width + 8));
        rect.setAttribute('height', String(box.height + 4));
        rect.setAttribute('rx', '3');
        label.insertBefore(rect, text);
      }

      if (state.selectedId === rel.id) {
        ['a', 'b'].forEach(function (key) {
          const handle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
          handle.setAttribute('class', 'rg-end-handle');
          handle.setAttribute('cx', String(path[key].x));
          handle.setAttribute('cy', String(path[key].y));
          handle.setAttribute('r', '6');
          handle.dataset.id = rel.id;
          handle.dataset.end = key;
          self.overlayLayer.appendChild(handle);
        });
      }
    });
    const rectBox = G.bboxOfRects(contentRects);
    const pointBox = G.bboxOfPoints(contentPoints);
    let box = rectBox;
    if (pointBox) {
      box = rectBox ? {
        x: Math.min(rectBox.x, pointBox.x),
        y: Math.min(rectBox.y, pointBox.y),
        width: Math.max(rectBox.x + rectBox.width, pointBox.x + pointBox.width) - Math.min(rectBox.x, pointBox.x),
        height: Math.max(rectBox.y + rectBox.height, pointBox.y + pointBox.height) - Math.min(rectBox.y, pointBox.y)
      } : pointBox;
    }
    this.updateSvgBounds(box);
  };

  CanvasView.prototype.ensureMarker = function (color) {
    const key = color.replace(/[^a-zA-Z0-9]/g, '');
    const id = 'rg-arrow-' + key;
    if (this.markers.has(id)) return id;
    const marker = document.createElementNS('http://www.w3.org/2000/svg', 'marker');
    marker.setAttribute('id', id);
    marker.setAttribute('markerWidth', '10');
    marker.setAttribute('markerHeight', '10');
    marker.setAttribute('refX', '9');
    marker.setAttribute('refY', '5');
    marker.setAttribute('orient', 'auto-start-reverse');
    marker.setAttribute('markerUnits', 'strokeWidth');
    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    path.setAttribute('d', 'M0,0 L10,5 L0,10 z');
    path.setAttribute('fill', color);
    marker.appendChild(path);
    this.defs.appendChild(marker);
    this.markers.set(id, true);
    return id;
  };

  CanvasView.prototype.updateSvgBounds = function (box) {
    const apply = function (svg, bx, by, bw, bh) {
      svg.setAttribute('viewBox', bx + ' ' + by + ' ' + bw + ' ' + bh);
      svg.style.left = bx + 'px';
      svg.style.top = by + 'px';
      svg.style.width = bw + 'px';
      svg.style.height = bh + 'px';
    };
    if (!box) {
      apply(this.svg, 0, 0, 1, 1);
      apply(this.overlaySvg, 0, 0, 1, 1);
      return;
    }
    const pad = 200;
    const bx = box.x - pad;
    const by = box.y - pad;
    const bw = box.width + pad * 2;
    const bh = box.height + pad * 2;
    apply(this.svg, bx, by, bw, bh);
    apply(this.overlaySvg, bx, by, bw, bh);
  };

  CanvasView.prototype.moveCardTo = function (id, x, y) {
    const card = this.nodeLayer.querySelector('.rg-card[data-id="' + id + '"]');
    if (!card) return;
    card.style.left = x + 'px';
    card.style.top = y + 'px';
  };

  CanvasView.prototype.updateEdges = function (doc, state) {
    this.edgeLayer.textContent = '';
    this.labelLayer.textContent = '';
    this.overlayLayer.textContent = '';
    this.renderEdges(doc, state || this.state);
  };

  CanvasView.prototype.screenToWorld = function (clientX, clientY) {
    return G.screenToWorld(clientX, clientY, this.viewport, this.el.getBoundingClientRect());
  };

  CanvasView.prototype.showRubberBand = function (d, color) {
    this.clearRubberBand();
    const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
    path.setAttribute('d', d);
    path.setAttribute('class', 'rg-rubber');
    path.setAttribute('stroke', color || '#5b8def');
    this.overlayLayer.appendChild(path);
  };

  CanvasView.prototype.clearRubberBand = function () {
    const old = this.overlayLayer.querySelector('.rg-rubber');
    if (old) old.remove();
  };

  CanvasView.prototype.showGuides = function (guide) {
    this.clearGuides();
    if (guide.guideX != null) {
      const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      line.setAttribute('class', 'rg-guide');
      line.setAttribute('x1', guide.guideX);
      line.setAttribute('x2', guide.guideX);
      line.setAttribute('y1', '-100000');
      line.setAttribute('y2', '100000');
      this.overlayLayer.appendChild(line);
    }
    if (guide.guideY != null) {
      const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      line.setAttribute('class', 'rg-guide');
      line.setAttribute('x1', '-100000');
      line.setAttribute('x2', '100000');
      line.setAttribute('y1', guide.guideY);
      line.setAttribute('y2', guide.guideY);
      this.overlayLayer.appendChild(line);
    }
  };

  CanvasView.prototype.clearGuides = function () {
    Array.from(this.overlayLayer.querySelectorAll('.rg-guide')).forEach(function (el) { el.remove(); });
  };

  CanvasView.prototype.setEmpty = function (text) {
    this.emptyEl.hidden = !text;
    this.emptyEl.textContent = text || '';
  };

  root.RGCanvasView = { CanvasView: CanvasView };
})(window);
```

```js
    armedId: null,
    collapsed: new Set(),
    history: [],
    histIndex: -1,
    view: null,
    drag: null,
    nameTimer: null

  function snapshot() {
    return JSON.stringify({ name: app.doc.name, entities: app.doc.entities, relations: app.doc.relations });
  }

  function restore(snap) {
    const data = JSON.parse(snap);
    app.doc.name = data.name;
    app.doc.entities = data.entities;
    app.doc.relations = data.relations;
    if (app.selectedId && !M.findObject(app.doc, app.selectedId)) app.selectedId = null;
    if (app.armedId && !M.findRelation(app.doc, app.armedId)) app.armedId = null;
  }

  function initHistory() {
    app.history = [snapshot()];
    app.histIndex = 0;
    updateHistoryButtons();
  }

  function pushHistory() {
    const snap = snapshot();
    if (snap === app.history[app.histIndex]) return;
    app.history = app.history.slice(0, app.histIndex + 1);
    app.history.push(snap);
    if (app.history.length > 100) app.history.shift();
    app.histIndex = app.history.length - 1;
    updateHistoryButtons();
  }

  function undo() {
    if (app.histIndex <= 0) return;
    app.histIndex -= 1;
    restore(app.history[app.histIndex]);
    renderAll();
    updateHistoryButtons();
  }

  function redo() {
    if (app.histIndex >= app.history.length - 1) return;
    app.histIndex += 1;
    restore(app.history[app.histIndex]);
    renderAll();
    updateHistoryButtons();
  }

  function updateHistoryButtons() {
    els.undo.disabled = app.histIndex <= 0;
    els.redo.disabled = app.histIndex >= app.history.length - 1;
  }

  function renderAll() {
    els.tree.innerHTML = RGTreeView.renderTreeHTML(app.doc, {
      selectedId: app.selectedId,
      armedId: app.armedId,
      collapsed: app.collapsed
    });
    app.view.render(app.doc, { selectedId: app.selectedId, armedId: app.armedId });
    updateEmpty();
    updateArmedChip();
  }

  function select(id) {
    app.selectedId = id;
    renderAll();
    if (id) centerOnObject(id);
  }

  function centerOnObject(id) {
    const obj = M.findObject(app.doc, id);
    if (!obj) return;
    let point = null;
    if (M.kindOf(obj) === 'entity' && obj.position) {
      const rect = app.view.getEntityRect(id);
      if (rect) point = { x: rect.x + rect.width / 2, y: rect.y + rect.height / 2 };
    }
    if (M.kindOf(obj) === 'relation' && obj.ends) {
      const a = app.view.resolveEnd(obj.ends.a);
      const b = app.view.resolveEnd(obj.ends.b);
      if (a && b) point = { x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 };
    }
    if (!point) return;
    const canvasRect = els.canvas.getBoundingClientRect();
    app.doc.viewport.x = canvasRect.width / 2 - point.x * app.doc.viewport.zoom;
    app.doc.viewport.y = canvasRect.height / 2 - point.y * app.doc.viewport.zoom;
    app.view.setViewport(app.doc.viewport);
  }

  function updateEmpty() {
    const total = app.doc.entities.length + app.doc.relations.length;
    const visible = app.doc.entities.some(function (e) { return !!e.position; }) ||
      app.doc.relations.some(function (r) { return M.relationVisibility(app.doc, r); });
    if (visible) app.view.setEmpty(null);
    else if (total === 0) app.view.setEmpty('在左栏新建根对象，或把左栏节点拖到这里');
    else app.view.setEmpty('把左栏节点拖到这里生成实例');
  }

  function updateArmedChip() {
    const rel = app.armedId ? M.findRelation(app.doc, app.armedId) : null;
    if (!rel) {
      app.armedId = null;
      els.armedGroup.hidden = true;
      return;
    }
    els.armedGroup.hidden = false;
    els.armedName.textContent = M.effectiveRelationLabel(app.doc, rel);
  }

  app.view = new RGCanvasView.CanvasView(els.canvas, {
    onCanvasPointerDown: function (e) {
      app.selectedId = null;
      renderAll();
    },
    onCardPointerDown: function (id, e) { startEntityMove(id, e); },
    onCardDoubleClick: function (id) { startRename(id); }
  });

  app.view.setViewport(app.doc.viewport);
  initHistory();
  renderAll();

  window.RGApp = { app: app, select: select, renderAll: renderAll, undo: undo, redo: redo };
})();
```

```js
  function toggleCollapse(id) {
    if (app.collapsed.has(id)) app.collapsed.delete(id);
    else app.collapsed.add(id);
    renderAll();
  }

  function addRoot(kind) {
    let obj;
    if (kind === 'entity') {
      obj = M.createEntity(app.doc, { name: M.nextEntityName(app.doc, null) });
    } else {
      obj = M.createRelation(app.doc, { name: M.nextRelationDefinitionName(app.doc, null) });
    }
    app.selectedId = obj.id;
    pushHistory();
    renderAll();
  }

  function addChild(parentId) {
    const parent = M.findObject(app.doc, parentId);
    if (!parent) return;
    let obj;
    if (M.kindOf(parent) === 'entity') {
      obj = M.createEntity(app.doc, {
        name: M.nextEntityName(app.doc, parent),
        prototypeId: parent.id
      });
    } else {
      obj = M.createRelation(app.doc, {
        name: M.nextRelationDefinitionName(app.doc, parent),
        prototypeId: parent.id
      });
    }
    app.selectedId = obj.id;
    pushHistory();
    renderAll();
  }

  function placeObject(obj) {
    const c = els.canvas.getBoundingClientRect();
    const world = app.view.screenToWorld(c.left + c.width / 2, c.top + c.height / 2);
    if (M.kindOf(obj) === 'entity') {
      M.placeEntity(obj, world.x - 84, world.y - 22);
    } else {
      M.placeRelation(obj, {
        a: M.makeFreeEnd(world.x - 70, world.y),
        b: M.makeFreeEnd(world.x + 70, world.y)
      });
    }
    pushHistory();
    renderAll();
  }

  function unplaceObject(obj) {
    if (M.kindOf(obj) === 'entity') M.unplaceEntity(obj);
    else M.unplaceRelation(obj);
    pushHistory();
    renderAll();
  }

  function openMenuAt(menu, x, y) {
    menu.hidden = false;
    menu.style.left = x + 'px';
    menu.style.top = y + 'px';
    setTimeout(function () {
      document.addEventListener('pointerdown', closeMenuOnOutside, { once: true });
    }, 0);
  }

  function closeMenuOnOutside(e) {
    if (!els.menu.contains(e.target)) closeMenu();
  }

  function closeMenu() {
    els.menu.hidden = true;
    els.menu.textContent = '';
  }

  function openRowMenu(anchor, id) {
    const obj = M.findObject(app.doc, id);
    if (!obj) return;
    const rect = anchor.getBoundingClientRect();
    els.menu.textContent = '';
    const addItem = function (text, fn, danger) {
      const btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'btn' + (danger ? ' danger' : '');
      btn.textContent = text;
      btn.addEventListener('click', function () {
        closeMenu();
        fn();
      });
      els.menu.appendChild(btn);
    };
    if (M.kindOf(obj) === 'entity') {
      addItem(obj.position ? '从画布移除' : '放到画布', function () {
        if (obj.position) unplaceObject(obj);
        else placeObject(obj);
      });
    } else {
      addItem(obj.ends ? '从画布移除' : '放到画布', function () {
        if (obj.ends) unplaceObject(obj);
        else placeObject(obj);
      });
      if (obj.ends) {
        addItem('反向', function () {
          const tmp = obj.ends.a;
          obj.ends.a = obj.ends.b;
          obj.ends.b = tmp;
          pushHistory();
          renderAll();
        });
      }
    }
    addItem('删除', function () {
      app.selectedId = id;
      deleteSelected();
    }, true);
    openMenuAt(els.menu, rect.left, rect.bottom + 4);
```

```js
  function nearestPort(entityId, clientX, clientY) {
    const rect = app.view.getEntityRect(entityId);
    if (!rect) return 'right';
    const ports = ['top', 'right', 'bottom', 'left'];
    let best = 'right';
    let bestDist = Infinity;
    const origin = app.view.el.getBoundingClientRect();
    ports.forEach(function (port) {
      const p = G.portPoint(rect, port);
      const screen = G.worldToScreen(p, app.doc.viewport, origin);
      const d = Math.hypot(screen.x - clientX, screen.y - clientY);
      if (d < bestDist) {
        bestDist = d;
        best = port;
      }
    });
    return best;
  }
```

```js
  const STORAGE_KEY = 'relation_graph_v1';
  let saveTimer = null;

  function scheduleSave() {
    clearTimeout(saveTimer);
    saveTimer = setTimeout(saveNow, 300);
  }

  function saveNow() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(app.doc));
    } catch (err) {
      toastr.warning('本地保存失败：' + err.message);
    }
  }

  function loadSaved() {
    let raw;
    try { raw = localStorage.getItem(STORAGE_KEY); } catch (err) { return null; }
    if (!raw) return null;
    let parsed;
    try { parsed = JSON.parse(raw); } catch (err) {
      backupBroken(raw);
      return null;
    }
    const result = M.validateDocument(parsed);
    if (!result.ok) {
      backupBroken(raw);
      toastr.warning('本地数据损坏，已备份并新建空图');
      return null;
    }
    return result.doc;
  }

  function backupBroken(raw) {
    try {
      localStorage.setItem(STORAGE_KEY + '_broken_' + Date.now(), raw);
    } catch (err) { /* ignore */ }
  }
```

```js
  function fitContent() {
    const rects = [];
    const points = [];
    app.doc.entities.forEach(function (e) {
      if (!e.position) return;
      const r = app.view.getEntityRect(e.id);
      if (r) rects.push(r);
    });
    app.doc.relations.forEach(function (r) {
      if (!M.relationVisibility(app.doc, r)) return;
      if (M.isEndFree(r.ends.a)) points.push(r.ends.a);
      if (M.isEndFree(r.ends.b)) points.push(r.ends.b);
    });
    const box = G.bboxOfRects(rects);
    const pointBox = G.bboxOfPoints(points);
    let merged = box;
    if (pointBox) {
      merged = box ? {
        x: Math.min(box.x, pointBox.x),
        y: Math.min(box.y, pointBox.y),
        width: Math.max(box.x + box.width, pointBox.x + pointBox.width) - Math.min(box.x, pointBox.x),
        height: Math.max(box.y + box.height, pointBox.y + pointBox.height) - Math.min(box.y, pointBox.y)
      } : pointBox;
    }
    const canvasRect = els.canvas.getBoundingClientRect();
    const vp = G.fitView(merged, { width: canvasRect.width, height: canvasRect.height }, 60);
    app.doc.viewport = vp;
```
