# 数学口算练习增强 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 `数学/math-quiz.html` 从「硬编码两位数加减」升级为「知识点注册表 + 年级难度档 + 两种题型 + 错题本/历史记录」的口算练习页。

**Architecture:** 题目统一表示为 token 流 `{ tokens, answerIndex, blankable }`——`tokens` 是表达式的展示片段数组，`answerIndex` 指向答案位，`blankable` 是可挖空的数值位。「求结果」= 输入框放在 answerIndex；「填空求未知数」= 把 answerIndex 挪到某个 blankable 位。同一份生成器因此同时支持两种题型。难度档只是「知识点 id 的集合」。学习闭环用 localStorage 三个键存历史/错题本/设置。

**Tech Stack:** 纯静态单文件 `数学/math-quiz.html`，vanilla JS（内联 `<script>`，无模块、无构建），内联 `<style>`。

function rand(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;

function pick(arr) {

// ── 出题 ──────────────────────────────────────
/** pool: 知识点 id 数组；types: { result, blank } —— 返回题目数组 */
function generateSession(pool, count, types) {
  const problems = [];
  for (let i = 0; i < count; i++) {
    const topic = pick(pool);
    const spec = TOPICS[topic].gen();
    let answerIndex = spec.answerIndex;
    if (types.blank && (!types.result || Math.random() < 0.5)) {
      answerIndex = pick(spec.blankable);
    }
    problems.push({ topic, tokens: spec.tokens, answerIndex });
  }
  return problems;
}

/** 题目显示文本（答案位显示为 □），用于错题本 */
function problemText(p) {
  return p.tokens.map((t, i) => (i === p.answerIndex ? '□' : t)).join(' ');
}

/** 题目正确答案 */
function problemAnswer(p) {
  return Number(p.tokens[p.answerIndex]);
}

// ── 批卷 ──────────────────────────────────────
function grade() {
  if (graded) return;

  let correctCount = 0;
  const wrongList = [];

  problems.forEach((p, i) => {
    const input = document.getElementById(`input-${i}`);
    const status = document.getElementById(`status-${i}`);
    const row = document.getElementById(`row-${i}`);
    const filled = input.value.trim() !== '';
    const val = Number(input.value);
    const isCorrect = filled && val === problemAnswer(p);

    if (isCorrect) {
      correctCount++;
      status.textContent = '✅';
      input.className = 'correct-answer';
      row.className = 'problem-row correct';
    } else {
      status.textContent = '❌';
      input.className = 'wrong-answer';
      row.className = 'problem-row wrong';
      wrongList.push({ index: i, problem: p, given: filled ? val : '未填' });
    }

    input.disabled = true;
  });

  graded = true;
  gradeBtn.disabled = true;
  scoreDisplay.textContent = correctCount * 10;

  showResult(correctCount, wrongList);
}

```js
() => {
  const bad = [];
  const toJs = (toks) => toks.join(' ').replace(/×/g, '*').replace(/÷/g, '/').replace(/−/g, '-');
  for (const [id, t] of Object.entries(TOPICS)) {
    for (let n = 0; n < 300; n++) {
      const s = t.gen();
      const ans = Number(s.tokens[s.answerIndex]);
      if (!Number.isFinite(ans)) { bad.push([id, 'NaN', s.tokens.join(' ')]); continue; }
      if (s.tokens.includes('…')) {
        // 有余数除法：被除数 = 除数 × 商 + 余数
        const [a, , b, , q, , r] = s.tokens;
        if (Number(a) !== Number(b) * Number(q) + Number(r)) bad.push([id, 'remainder', s.tokens.join(' ')]);
        if (ans !== Number(q)) bad.push([id, 'answerIndex!=quotient', s.tokens.join(' ')]);
      } else {
        const eqIdx = s.tokens.indexOf('=');
        const lhs = toJs(s.tokens.slice(0, eqIdx));   // '=' 之前的表达式
        const rhs = toJs(s.tokens.slice(eqIdx + 1));  // '=' 之后的结果
        const v = eval(lhs);
        if (v !== Number(rhs)) bad.push([id, 'lhs!=rhs', s.tokens.join(' '), v, rhs]);
        if (Number(rhs) !== ans) bad.push([id, 'answerIndex wrong', s.tokens.join(' '), rhs, ans]);
      }
      if (!/^\d+$/.test(s.tokens[s.answerIndex])) bad.push([id, 'answer not numeric', s.tokens.join(' ')]);
      for (const bi of s.blankable) {
        if (bi === s.answerIndex) bad.push([id, 'blankable overlaps answer', s.tokens.join(' ')]);
        if (!/^\d+$/.test(s.tokens[bi])) bad.push([id, 'blankable not numeric', s.tokens.join(' ')]);
      }
    }
  }
  return bad;
}

const STORE = {
  read(key, fallback) {
    try {
      const raw = localStorage.getItem(key);
      return raw ? JSON.parse(raw) : fallback;
    } catch (e) {
      return fallback;
    }

  write(key, val) {

function renderTopicsRow() {

// ── 错题本 ────────────────────────────────────
/** 错题去重键：token 流 + 挖空位置 */
function wrongKey(p) {
  return p.tokens.join('|') + '@' + p.answerIndex;
}

function readWrongBook() {
  const book = STORE.read(K.wrong, {});
  return book && typeof book === 'object' ? book : {};
}

/** 错题最集中的知识点名（无错题返回空串） */
function weakestTopic(book) {
  let best = '';
  let bestN = 0;
  for (const [topic, list] of Object.entries(book)) {
    const n = list.reduce((s, r) => s + r.count, 0);
    if (n > bestN) { bestN = n; best = topic; }
  }
  return best ? (TOPICS[best] ? TOPICS[best].name : '') : '';
}

/** 批卷后：写历史 + 更新错题本 + 算统计 */
function onGraded(correctCount, wrongList) {
  const ts = Date.now();
  const duration = Math.round((ts - startTime) / 1000);

  const history = STORE.read(K.history, []);
  const hist = Array.isArray(history) ? history : [];
  hist.push({ ts, level: levelSelect.value, count: problems.length, correct: correctCount, duration });
  while (hist.length > 100) hist.shift();
  STORE.write(K.history, hist);

  const book = readWrongBook();
  const wrongSet = new Set(wrongList.map(w => wrongKey(w.problem)));
  problems.forEach(p => {
    const list = book[p.topic] || (book[p.topic] = []);
    const k = wrongKey(p);
    const idx = list.findIndex(r => r.key === k);
    if (wrongSet.has(k)) {
      if (idx >= 0) list[idx].count = Math.min(list[idx].count + 1, 5);
      else list.push({ key: k, tokens: p.tokens, answerIndex: p.answerIndex, count: 1, ts });
    } else if (isRedo && idx >= 0) {
      list[idx].count -= 1;
      if (list[idx].count <= 0) list.splice(idx, 1);
    }
    if (!list.length) delete book[p.topic];
  });
  STORE.write(K.wrong, book);

  const totCorrect = hist.reduce((s, h) => s + h.correct, 0);
  const totCount = hist.reduce((s, h) => s + h.count, 0);
  lastStats = {
    duration,
    rate: totCount ? Math.round((totCorrect / totCount) * 100) : 0,
    weakest: weakestTopic(book),
  };
}

function renderWrongList() {
  const book = readWrongBook();
  const groups = Object.entries(book).filter(([, list]) => list.length);
  if (!groups.length) {
    wrongList.innerHTML = '<div class="none">还没有错题，真棒！🎉</div>';
    return;
  }
  wrongList.innerHTML = groups.map(([topic, list]) => {
    const head = `<div class="wrong-group-head">${TOPICS[topic] ? TOPICS[topic].name : topic}（${list.length}）</div>`;
    const items = list.map(r =>
      `<div class="wrong-item"><span>${r.tokens.map((t, i) => (i === r.answerIndex ? '□' : t)).join(' ')} → <strong>${Number(r.tokens[r.answerIndex])}</strong></span><span class="cnt">错 ${r.count} 次</span></div>`
    ).join('');
    return head + items;
  }).join('');
}

/** 用错题本里的题重开一轮（最多 30 题） */
function redoWrong() {
  const book = readWrongBook();
  const items = Object.entries(book).flatMap(([topic, list]) =>
    list.map(r => ({ topic, tokens: r.tokens, answerIndex: r.answerIndex }))
  );
  if (!items.length) return;

  problems = items.slice(0, 30);
  graded = false;
  isRedo = true;
  startTime = Date.now();
  wrongOverlay.classList.remove('show');
  overlay.classList.remove('show');
  confettiContainer.innerHTML = '';
  scoreDisplay.textContent = '—';
  scoreTotal.textContent = problems.length * 10;
  render();
  subtitle.textContent = `错题重做 · 一共 ${problems.length} 题`;
  document.getElementById('input-0')?.focus();
}
```
